         const rows = await query(

              COALESCE(jsonb_agg(jsonb_build_object('nombre', c.nombre, 'slug', c.slug, 'relevancia', oc.relevancia)), '[]'::jsonb)::text as categorias
            FROM embeddings_cache ec

            GROUP BY ec.ordenanza_id, ec.vector, o.numero, o.anio, o.titulo, o.resumen, o.estado`,
           [model]
         );
         return rows;
       }

    **Install dependencies first:** `npm install p-limit js-tiktoken`

    **CONFIG block:**
    ```typescript
    const CONFIG = {
      MODEL: "text-embedding-3-large" as const,
      CONCURRENCY: parseInt(process.env.EMBEDDING_CONCURRENCY ?? "3", 10),
      BATCH_SIZE: parseInt(process.env.EMBEDDING_BATCH_SIZE ?? "100", 10),  // ordinances per DB page
      TOKEN_THRESHOLD: 6000,  // tokens; above this use resumen
      MIN_TEXT_LENGTH: 100,   // chars; below this skip or use resumen
      API_MAX_RETRIES: 5,
      API_TIMEOUT: 60_000,   // 60s per request
      DELAY_MS: 200,         // delay between batches
    };
    ```

    **Script structure:**

    1. **OpenAI client:** Separate instance with `maxRetries: 5`, `timeout: 60_000`. NOT reusing MCP server's client.

    2. **DB connection:** Use `pg.Pool` directly (same as `concurrent-deepseek.ts` pattern — import from `../db/index.js` or create own pool from `DATABASE_URL`).

    3. **Token counting:** Use `js-tiktoken` with `cl100k_base` encoder:
       ```typescript
       import { encodingForModel } from "js-tiktoken";
       // Use "text-embedding-3-large" compatible encoding
       const enc = encodingForModel("text-embedding-3-large");
       const tokenCount = enc.encode(text).length;
       ```

    4. **Text selection logic:**
       ```typescript
       function selectText(ordenanza: { texto_completo: string | null; resumen: string | null; palabras_clave: string[] }): string | null {
         const fullText = ordenanza.texto_completo?.trim() ?? "";
         const resumen = ordenanza.resumen?.trim() ?? "";

         // Prefer full text if long enough and within token limit
         if (fullText.length >= CONFIG.MIN_TEXT_LENGTH) {
           const tokens = enc.encode(fullText).length;
           if (tokens <= CONFIG.TOKEN_THRESHOLD) return fullText;
         }

         // Fall back to resumen
         if (resumen.length >= CONFIG.MIN_TEXT_LENGTH) return resumen;

         // Both too short — skip
         return null;
       }
       ```

    5. **Main flow:**
       ```
       async function main():
         a. Query unembedded ordinances:
            SELECT o.id, o.texto_completo, o.resumen, o.palabras_clave
            FROM ordenanzas o
            WHERE o.id NOT IN (
              SELECT ordenanza_id FROM embeddings_cache WHERE modelo = 'text-embedding-3-large'
            )
            AND o.texto_completo IS NOT NULL
            ORDER BY o.anio DESC, o.numero DESC

         b. Log: "Found N ordinances to embed"

         c. Process with p-limit(CONFIG.CONCURRENCY):
            For each batch of CONFIG.BATCH_SIZE:
              - Select text for each ordinance
              - Skip if null (log as "skipped")
              - Call openai.embeddings.create({ model: MODEL, input: texts })
              - Batch INSERT into embeddings_cache:
                INSERT INTO embeddings_cache (ordenanza_id, modelo, vector, dimensions)
                VALUES ($1, 'text-embedding-3-large', $2, '3072')
                ON CONFLICT (ordenanza_id, modelo) DO UPDATE SET
                  vector = EXCLUDED.vector, updated_at = NOW()
              - Progress log: "Processed M/N (X%)"
              - await delay(CONFIG.DELAY_MS)

         d. Summary: total, embedded, skipped, errors, elapsed time
         e. process.exit(0)
       ```

    6. **Error handling:**
       - Individual API errors: log and skip, continue
       - 429 rate limit: log "Rate limited, waiting..." and sleep 10s, then retry once
       - Fatal errors (DB connection): log and process.exit(1)

    7. **Run command:** `npx tsx src/processor/generate-embeddings.ts`
